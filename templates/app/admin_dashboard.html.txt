                                submitButton.innerText = 'Submitting...';  // Change text to "Submitting..."
                        
                                var formData = new FormData(this);  // Get form data
                        
                                fetch("{% url 'add_toddler' %}", {
                                    method: "POST",
                                    body: formData,
                                })
                                .then(response => response.json())
                                .then(data => {
                                    var resultDiv = document.getElementById("add-toddler-result");
                                    if (data.success) {
                                        resultDiv.innerHTML = '<div class="alert alert-success">Toddler added successfully!</div>';
                                        document.getElementById('add-toddler-form').reset();
                                    } else {
                                        resultDiv.innerHTML = '<div class="alert alert-danger">' + data.error + '</div>';
                                    }
                        
                                    submitButton.disabled = false;
                                    submitButton.innerText = 'Register';  // Reset button text
                                })
                                .catch(error => {
                                    console.error('Error:', error);
                                    document.getElementById("add-toddler-result").innerHTML = '<div class="alert alert-danger">An error occurred. Please try again later.</div>';
                        
                                    submitButton.disabled = false;
                                    submitButton.innerText = 'Register';  // Reset button text
